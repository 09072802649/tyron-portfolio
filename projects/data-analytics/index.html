SQL Aggregation Logic

-- Task 6&7 Program Performance Analysis (BSIT, BSIS, BSCS)
SELECT
Program,
COUNT(*) AS Number_of_Students,
AVG(Study_Hours) AS Average_Study_Hours,
AVG(Final_Grade) AS Average_Final_Grade
FROM
student_academic_records
GROUP BY
Program
ORDER BY
Average_Final_Grade DESC;
-- Task 3 Descriptive Statistics for Final Grades
SELECT
AVG(Final_Grade) AS Mean,
MIN(Final_Grade) AS Minimum,
MAX(Final_Grade) AS Maximum
FROM
student_academic_records;
-- Task 5 Pass/Fail Count and Overall Passing Percentage
SELECT
Passed,
COUNT(*) AS Student_Count,
ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM student_academic_records), 2) AS Percentage
FROM
student_academic_records
GROUP BY
Passed;
